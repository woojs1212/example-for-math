    function shuffle(arr) {

      const copy = [...arr];

      for (let i = copy.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [copy[i], copy[j]] = [copy[j], copy[i]];

      }

      return copy;

    }

    function createAnalogClockSvg(hour, minute) {

      const minuteAngle = minute * 6;

      const hourAngle = ((hour % 12) + minute / 60) * 30;


      let ticksHtml = '';

      for (let i = 0; i < 60; i++) {

        const angle = i * 6;

        const isHourMark = i % 5 === 0;

        const y1 = isHourMark ? 14 : 16;

        const y2 = isHourMark ? 25 : 21;

        const strokeW = isHourMark ? 3.2 : 1.3;

        const color = isHourMark ? '#1E293B' : '#94A3B8';

        ticksHtml += `<line x1="100" y1="${y1}" x2="100" y2="${y2}" stroke="${color}" stroke-width="${strokeW}" stroke-linecap="round" transform="rotate(${angle} 100 100)"/>`;

      }


      let numsHtml = '';

      for (let n = 1; n <= 12; n++) {

        const rad = ((n * 30 - 90) * Math.PI) / 180;

        const x = 100 + 62 * Math.cos(rad);

        const y = 100 + 62 * Math.sin(rad);

        numsHtml += `<text x="${x}" y="${y}" text-anchor="middle" dominant-baseline="central" font-size="16" font-weight="900" fill="#1E293B">${n}</text>`;

      }


      return `

        <svg viewBox="0 0 200 200" class="w-full h-full drop-shadow-md animate-pop">
